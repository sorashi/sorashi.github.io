Algorithm Pseudocode Splitters_by_Rank(S, rank)
void Splitters_by_Rank(int** S, int rank) {
int L[p], R[p], m[p]; int v;
for (i = 0; i < p; i++) { L[i] = 0; R[i] = n/p-1; }
while (there exists i such that L[i] < R[i]) {
v = Pickup_Random_Pivot(S, L, R);
for (i = 0; i < p; i++)
m[i] = binarySearch(v, S[i]);
if (m[0] + .. + m[p-1] >= rank)
R[0..p-1] = m[0..p-1]; // narrow bounds from right
else
L[0..p-1] = m[0..p-1]; // narrow bounds from left
}
return L[0..p-1];
}
